# Use a proxy via SSH tunnel
# https://superuser.com/questions/819714/chrome-ssh-tunnel-forwarding
# -N: don't run a command: just make redirections/proxies
# -D: set up a proxy server on local port 1080
ssh -vND 4567 remote_server
google-chrome --incognito --proxy-server="socks5://localhost:4567"

# Open a window in a specific profile (works whether Chrome is running or not)
# Directory names (Default, Profile 1, ...) differ from display names:
# see "profile"."info_cache" in ~/.config/google-chrome/"Local State"
google-chrome --profile-directory="Profile 1"

# Add per-profile entries to the taskbar/dock right-click menu (e.g. Dash to Panel)
# Copy the desktop file to ~/.local/share/applications (shadows the system one),
# then add an action per profile. Delete the copy to restore defaults.
cp /usr/share/applications/google-chrome.desktop ~/.local/share/applications/
# In the copy: append to Actions= line, e.g. Actions=new-window;profile-work;
# then add a matching section:
#   [Desktop Action profile-work]
#   Name=New Window — Work
#   Exec=/usr/bin/google-chrome-stable --profile-directory="Profile 1"
desktop-file-validate ~/.local/share/applications/google-chrome.desktop
# Those Name= lines are static: renaming a profile in Chrome leaves them stale, so
# the menu drifts from the profile picker. After a rename, list the current
# display name per profile directory and update the Name= lines to match:
python3 -c 'import json,pathlib; d=json.loads((pathlib.Path.home()/".config/google-chrome/Local State").read_text()); [print(k,"=",v["name"]) for k,v in d["profile"]["info_cache"].items()]'
